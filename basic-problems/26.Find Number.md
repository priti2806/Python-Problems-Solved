Write a python program to search a given number from a list

nums = [10, 20, 30, 40, 50]

search = int(input("Enter number to search: "))

found = False

for i in range(len(nums)):
    if nums[i] == search:
        print("Number found at index", i)
        found = True
        break

if not found:
    print("Number not found")
