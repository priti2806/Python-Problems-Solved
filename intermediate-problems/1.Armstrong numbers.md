
Print all the armstrong numbers in the range of 100 to 1000--

def armstrong_number():
    result=[]
    for i in range(100,1000):
        power = len(str(i))
        sum = 0
        for j in str(i):
            sum += int(j)**power
        if sum == i:
            result.append(i)
    return result
print(armstrong_number())
